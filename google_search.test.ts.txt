import { test, expect } from '@playwright/test';

test('should search for Playwright on Google', async ({ page }) => {
  // 1. Open Google
  await page.goto('https://www.google.com');

  // 2. Type "Playwright" into the search box and press Enter
  const searchInput = page.locator('textarea[name="q"], input[name="q"]');
  await searchInput.fill('Playwright');
  await searchInput.press('Enter');

  // 3. Make sure the results page loaded
  await expect(page).toHaveTitle(/Playwright/);
});
